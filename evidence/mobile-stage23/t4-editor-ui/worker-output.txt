const { createWriteStream, fstatSync } = require("node:fs");
const { Writable } = require("node:stream");
if (process.env.NODE_TEST_CONTEXT === "child-v8") {
  for (const [name, fd] of [["stdout", 1], ["stderr", 2]]) {
    if (process[name].constructor === Writable && fstatSync(fd).isSocket()) {
      Object.defineProperty(process, name, { value: createWriteStream(null, { fd, autoClose: false }), configurable: true });
    }
  }
}
